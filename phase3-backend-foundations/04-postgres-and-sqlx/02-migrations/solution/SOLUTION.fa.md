pub async fn run_migrations(pool: &PgPool) -> Result<(), sqlx::migrate::MigrateError> {
    sqlx::migrate!("./migrations").run(pool).await
}

pub async fn insert_widget(pool: &PgPool, name: &str) -> Result<i64, sqlx::Error> {
    sqlx::query_scalar("INSERT INTO p3_04_02_widgets (name) VALUES ($1) RETURNING id")
        .bind(name)
        .fetch_one(pool)
        .await
}

pub async fn count_widgets(pool: &PgPool) -> Result<i64, sqlx::Error> {
    sqlx::query_scalar("SELECT COUNT(*) FROM p3_04_02_widgets")
        .fetch_one(pool)
        .await
}

## چرا تست اول از همه وضعیت دیتابیس رو به طور دستی ریست می‌کنه

دیتابیسِ این درس (`p3_04_02_migrations`) بین اجراهای مختلفِ تست‌ها (test runs) **هرگز پاک نمی‌شه (wiped)** — برخلافِ یه `HashMap` که تو مموری بود و با هر بار اجرای `cargo test` کلاً یه نسخه‌ی تر و تازه‌اش متولد می‌شد، وضعیتِ دیتابیس Postgres بین اجراهای مختلف ماندگاره (persists). اگه به خاطر اجرایِ *قبلیِ* این تست جدول `_sqlx_migrations` ثبت کرده باشه که مایگریشن `0001` قبلاً اعمال شده، اونوقت تو اجرایِ دومِ تست، اون ادعایی که می‌کردیم "جدول وجود داره و بلافاصله بعد از مایگریشن کاملاً خالیه" کاملاً بی‌معنی می‌شه — چون ممکنه اون جدول هنوزم حاویِ ردیف‌هایی باشه که از اجرایِ قبلیِ تستِ `inserted_widgets_are_counted` جا موندن. پاک کردن جدول و حذفِ ردیف رهگیریِ اون تو خط اول کُد، باعث می‌شه که تست واقعاً با یه دیتابیس تمیز (clean slate) روبه‌رو بشه تا عملیات مایگریشنش رو روش انجام بده، تا بتونه واقعاً درستیِ این جمله رو تست کنه که "عملیات مایگریشن، یه جدول خالی می‌سازه" نه اینکه فقط فرض کنه که این اتفاق می‌افته.

## چرا این درس دیتابیس اختصاصی خودش رو داره

هر درسِ دیگه‌ای که با Postgres سر و کار داره، از همون دیتابیسِ مشترکِ `taskforge` استفاده می‌کنه و صرفاً با پیشوند زدن به اسمِ *جداول* (`p3_04_02_widgets`، `p3_04_03_anime`، یا `jobs` تو `taskforge-storage`) از تداخل جلوگیری می‌کنه. اما این ترفند برای جدولِ دفتریِ خودِ `!sqlx::migrate` یعنی `_sqlx_migrations` جوابگو نیست — این جدول اصلاً برایِ هر کریت فضای نام مجزا (namespaced) نداره، بلکه تو هر *دیتابیس* فقط یه دونه ازش هست، و کلیدش هم فقط نسخه‌یِ عددیِ هر مایگریشونه. فایلِ `0001_create_widgets.sql` برای این درس و فایلِ `0001_create_jobs.sql` برای کریت `taskforge-storage` هر دوشون "نسخه‌ی 1" هستن؛ اگه قرار باشه ماکرویِ `!sqlx::migrate` تو هر دو کریت روی یه دیتابیس یکسان ران بشه، اونی که دوم از همه اجرا بشه می‌بینه که نسخه‌ی 1 قبلاً با چک‌سامِ یه فایل *دیگه* ثبت شده و با خطای `VersionMismatch` متوقف می‌شه — و بدتر از اون، تستِ `migrating_creates_the_widgets_table` تو همین درس که برای داشتنِ یه شروع تمیز میاد و ردیفِ رهگیریِ نسخه‌ی 1 رو پاک می‌کنه، باعث می‌شه که در کمال سکوت و نامحسوس، رهگیریِ مایگریشن‌های مربوط به کریتِ دیگه کلاً نابود بشن و به هم بریزن (corrupt بشن). این یه حالت لبه و فرضی (hypothetical edge case) نیست که فقط واسه فایل README تراشیده باشنش — این دقیقاً همون تداخلیه که تو مجموعه‌تستِ (test suite) همین مخزن و زمانی که این دو تا درس داشتن به طور همزمان روی همون دیتابیس مشترک تست می‌شدن، رخ داد، و دقیقاً همون دلیلیه که به خاطرش این درس به یه دیتابیسِ اختصاصیِ خودش یعنی `p3_04_02_migrations` منتقل شد.

## چرا تمام تست‌ها به برچسب `[(serial(p3_04_02_migrations_db#]` نیاز دارن

رفتارِ پیش‌فرضِ `cargo test` — که تک‌تکِ توابعِ `[test]#` رو به صورت هم‌زمان (concurrently) و روی ریسمان‌های مجزا (separate threads) ران می‌کنه — برای حجمِ عظیمی از تست‌های تو این مخزن کاملاً ایمن (safe) هستش دقیقاً به این خاطر که اکثرشون هیچ چیزی رو به جز وضعیتِ داخل‌حافظه‌ایِ (in-memory) تازه ساخته‌شده‌ی خودشون، لمس نمی‌کنن. اما این درس استثناست: هر سه تا تست به یک دیتابیسِ زنده‌ی یکسانِ Postgres متصل می‌شن (hit)، و تستِ `migrating_creates_the_widgets_table` به طور کاملاً فعالی میاد و همون جدولی رو که اون دوتایِ دیگه بهش وابسته‌ان، از بیخ پاک می‌کنه و دوباره می‌سازه (drops and recreates). اگه صفتِ `[serial]#` رو نذاری، موقعِ ساخته شدن همین مخزن دقیقاً همین مشکل پیش اومد — تستِ `inserted_widgets_are_counted` یه وقتایی در کمال تعجب fail می‌شد، اونم فقط وقتی که تست‌ها به طور موازی اجرا می‌شدن و هرگز با زدنِ `1=test-threads--` مشکلی نداشت — چون تستِ `migrating_creates_the_widgets_table` که داشت همزمان ران می‌شد، می‌تونست دقیقاً تو اون نقطه‌ی زمانیِ بینِ دوتا محاسبه‌ای که تست اول داشت واسه شمارشِ "قبل" (before) و "بعد" (after) می‌کرد، بیاد و کلاً جدول رو دراپ کنه. برچسبِ `[(serial(p3_04_02_migrations_db#]` همه‌ی تست‌هایی رو که دقیقاً همون تگِ یکسان رو با خودشون دارن، مجبور می‌کنه که فقط و فقط یکی‌یکی ران بشن، اونم تو هر ترتیبی که خودشون می‌خوان ولی به شرطی که هیچ‌وقت لای‌به‌لای هم (interleaved) قرار نگیرن — و با این کار، یه مشکل که "احتمال داشت با توجه به وضعیت زمان‌بندیِ ریسمان‌ها گاهی خراب بشه (flaky)" رو به وضعیتِ "تو هر دفعه اجرا تضمین شده است که درسته" تبدیل می‌کنه، بدون اینکه لازم باشه کلاً بزنی `1=test-threads--` (که البته اگه تستِ دیگه‌ای هم تو همون باینری بود الکی اونا رو هم می‌کرد تو صف و ترتیبی‌شون می‌کرد).

## اضافه کردن `0002` در مقابل ادیت کردنِ `0001` بعد از اجرا شدنشون

اضافه کردن فایلِ `0002_add_widget_price.sql` و ران کردنِ تابع `run_migrations` روی دیتابیسی که قبلاً `0001` روش اِعمال شده: این دقیقاً همون فرآیند و روالیه (workflow) که ازش انتظار می‌ره — `0001` به کلی نادیده گرفته می‌شه (چون قبلاً ثبت شده)، و `0002` ران می‌شه و ثبت می‌شه. این در واقع همون روش عادی و ایمنیه (safe way) که با استفاده ازش، شِماها (schemas) تو طول زمان تکامل (evolve) پیدا می‌کنن.

اما، دست‌کاری کردن (Editing) فایلِ `0001_create_widgets.sql` *بعد از اینکه اِعمال شده*، واقعاً یه تله و خودزنیِ خطرناک (footgun) هست: کتابخونه‌ی `sqlx` به همراهِ نسخه‌ی هر فایلِ مایگریشن، یه چک‌سام (checksum) از محتوای اون فایل هم تو دیتابیس ثبت می‌کنه. در نتیجه تو فراخوانیِ بعدیِ `run_migrations`، وقتی بره سراغ دیتابیسی که نسخه‌ی اولیه‌ی `0001` رو اعمال کرده بود، ابزار `sqlx` متوجه عدم تطابقِ (mismatch) چک‌سام می‌شه و کلاً با یه ارورِ `VersionMismatch` / چک‌سام، متوقف می‌شه — اونم کاملاً عامدانه، چون اجرای مجدد یه مایگریشنِ تغییرکرده رو دیتابیسی که قبلتر نسخه‌ی *قدیمیش* رو گرفته (یا نادیده گرفتنِ اون تغییر تو سکوت کامل)، دقیقاً همون نقطه‌ایه که تاریخچه‌ی مایگریشن و وضعیتِ واقعیِ دیتابیس مسیرشون از هم جدا می‌شه و با هم تناقض (drift apart) پیدا می‌کنن. راه‌حل درست و اصولی برای اینکه "من نیاز دارم تا یه چیزی رو که قبلاً مایگریت شده بود رو تغییر بدم" همیشه و همیشه اینه که یه فایل مایگریشن *جدید* بنویسی که شمای موجود رو تغییر بده (alters)، نه اینکه بری و کدهای تاریخچه‌ای رو از بیخ ادیت کنی — این دقیقاً همون دیسیپلین و نظمیه که جنگو هم یادت می‌داد وقتی مجبورت می‌کرد تا به جای ویرایشِ دستیِ فایلِ مایگریشن قبلی، دوباره `makemigrations` رو بزنی.

## چرا تو مرحله‌ی کامپایل تعبیه‌شدنِ مایگریشن‌ها برای دیپلوی (deployment) مهمه

چون کدهای SQL مربوط به مایگریشن‌ها مستقیماً تو فایلِ باینری کامپایل‌شده‌ی نهایی جاسازی (baked) می‌شن (ماکرویِ `!sqlx::migrate` اونا رو تو زمانِ کامپایل با کمک یه ماکروی پروسیجرال (proc macro) می‌خونه)، یه آرتیفکت که واسه ریلیز دیپلوی می‌شه کلاً تو خودش کامل و بی‌نیازه (fully self-contained) — یعنی فقط همون یه دونه فایل باینری رو روی سرور کپی کن و ران کن، تا موقع استارت شدن خودش دیتابیس خودشو مایگریت کنه، بدون اینکه نیاز باشه یه دایرکتوریِ `/migrations` جداگانه به همراهش جابجا بشه، و هیچ خطری هم نیست که فایلِ باینری و فایل‌های مایگریشن با هم عدم تطابق پیدا کنن چون یکی یادش رفته باشه یه کپی ازشون بگیره. این رو مقایسه کن با خوندنِ فایل‌های `sql.` تو زمان اجرا (runtime) از روی دیسک، که نیازمند این بود که در کنار هر نسخه‌ی دیپلوی‌شده، یه دایرکتوریِ اضافی رو جابجا (shipping) کنی و حواست هم باشه که دقیقاً جای درستی قرار بگیره.

## تو مقایسه با جدول `django_migrations` جنگو

از لحاظ ساختاری کاملاً یه ایده‌ی مشابهن: یه جدول برای رهگیری که ثبت می‌کنه کدوم مایگریشن‌ها اعمال شدن، و این رو قبل از ران کردن مایگریشن‌های جدید چک می‌کنه. تفاوت اصلی تو اینه که *محتوایِ* مایگریشن‌ها از کجا میاد. ابزارِ `makemigrations` تو جنگو تغییراتِ کلاس‌های مُدل‌های (model classes) تو رو با آخرین وضعیتِ شناخته‌شده از شِما (schema) مقایسه (diff) می‌کنه و *به صورت خودکار* برات کدهای پایتون/SQLِ مایگریشن رو *تولید (generates)* می‌کنه؛ ابزارِ `sqlx` کلاً هیچ مدلی نداره که بخواد روش عملِ دیف گرفتن رو انجام بده — خودت با دستِ خودت باید کدهای SQLِ هر مایگریشن رو بنویسی، و ماکروی `!sqlx::migrate` فقط مسئولیتِ اعمال‌کردن و رهگیریشون (applying/tracking) رو به عهده می‌گیره، نه تولید کردنشون. این یه نتیجه‌ی مستقیم از این واقعیته که کلاً تو محیطِ Rust/`sqlx` اصلاً هیچ ORMی در کار نیست که مدل‌های ذخیره‌شده رو تو زمان اجرا (runtime model registry) داشته باشه و بتونه باهاشون دیف بگیره — که این در واقع بهای فلسفه‌یِ `sqlx` (یعنی «خودت کدهای واقعی SQL می‌نویسی، در عوض ما برات کامل و از لحاظ نوع، چکشون می‌کنیم») در مقابلِ فلسفه‌ی ORM جنگو («فریم‌ورک کدهای SQL رو واسه تو می‌نویسه») هستش.

## تابع `query_scalar` در برابر `query_as`

تابعِ `query_scalar` مشخصاً دقیقاً برای زمان‌هایی مناسبه که یه کوئری **دقیقاً و فقط یه ستون (one column)** به ازای هر ردیف برمی‌گردونه — این متد مستقیماً همون یه دونه ستون رو رمزگشایی (decodes) می‌کنه و می‌ذاردش تو یه متغیرِ ساده‌ی Rust (که اینجا `i64` بود)، و کلاً از مرحله‌ی رمزگشایی کردن تو قالب یه ردیف/ساختار (row/struct) که مرحله‌ی واسطه‌ست چشم‌پوشی می‌کنه. از اون طرف تابعِ `query_as` برای کوئری‌هایی هستش که **چندین ستون (multiple columns)** برمی‌گردونن و قرارشون هم اینه که به یه ساختار (یا توپل) که خصیصه‌ی `FromRow` رو پیاده‌سازی کرده نقشه‌بندی (map) بشن — یعنی دقیقاً همون کاری که کریت `taskforge-storage` با نوعِ `JobRow` واسه ردیف‌های جدولِ چند-ستونه‌ی `jobs` انجام می‌ده. اگه تو یه کوئریِ چند-ستونه بخوای از `query_scalar` استفاده کنی اصلاً خطایِ کامپایل می‌گیری (چون دیتای `SELECT id, name, created_at` اصلاً نمی‌تونه تو یه نوعِ اسکالر و ساده‌ی تکی جا بگیره)، و استفاده کردن از `<_, i64>::query_as` رویِ خروجیِ `(*)SELECT COUNT` هم از نظرِ فنی کار می‌کنه و جواب می‌ده اما خب در مقایسه با تابعِ سفارشیِ (purpose-built) `query_scalar`، واقعاً به شکل بی‌خود و بی‌جهتی پیچیده و غیرمستقیم به نظر میاد.
